cmake_minimum_required(VERSION 3.16)
project(quant_orderbook VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(LOB_NATIVE "Tune binaries for this machine (not portable)" OFF)
option(LOB_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)
option(LOB_BUILD_BENCHMARKS "Build x86 timestamp-counter benchmarks" ON)
find_package(Threads REQUIRED)
add_library(lob INTERFACE)
target_include_directories(lob INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(lob INTERFACE Threads::Threads)
if(MSVC)
    target_compile_options(lob INTERFACE /W4 /permissive-)
    target_compile_definitions(lob INTERFACE NOMINMAX)
else()
    target_compile_options(lob INTERFACE -Wall -Wextra)
    if(LOB_NATIVE)
        target_compile_options(lob INTERFACE -march=native)
    endif()
    if(MINGW)
        target_link_options(lob INTERFACE -static)
    endif()
endif()
if(LOB_SANITIZERS)
    if(MSVC OR WIN32)
        message(FATAL_ERROR "Use GCC/Clang on Linux for LOB_SANITIZERS")
    endif()
    target_compile_options(lob INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(lob INTERFACE -fsanitize=address,undefined)
endif()
function(lob_executable name source)
    add_executable(${name} ${source})
    target_link_libraries(${name} PRIVATE lob)
endfunction()
enable_testing()
lob_executable(test_correctness tests/test_correctness.cpp)
add_test(NAME correctness COMMAND test_correctness)
lob_executable(test_feed tests/test_feed.cpp)
add_test(NAME feed COMMAND test_feed)
lob_executable(test_flat_hash_boundaries tests/test_flat_hash_boundaries.cpp)
add_test(NAME flat_hash_boundaries COMMAND test_flat_hash_boundaries)
lob_executable(test_engine_boundaries tests/test_engine_boundaries.cpp)
add_test(NAME engine_boundaries COMMAND test_engine_boundaries)
lob_executable(replay_events tools/replay_events.cpp)
if(LOB_BUILD_BENCHMARKS)
    foreach(name rdtsc itch_pipeline zero_copy trading_loop risk journal flat_hash signals advanced_orders)
        lob_executable(bench_${name} bench/bench_${name}.cpp)
    endforeach()
endif()
